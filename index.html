תנאים: if, else if, else

תנאי מאפשר לקוד לבחור מסלול פעולה לפי שאלה שהתשובה שלה היא true או false.

const grade = 82;

if (grade >= 90) {
    console.log("מצוין");
} else if (grade >= 75) {
    console.log("טוב מאוד");
} else if (grade >= 60) {
    console.log("עבר");
} else {
    console.log("צריך תרגול נוסף");
}

הבדיקה נעצרת בתנאי הראשון שמתקיים. לכן חשוב לסדר את התנאים מהספציפי לכללי.

לוגיקה בוליאנית: &&, ||, !

לוגיקה בוליאנית מאפשרת לשלב כמה שאלות לתנאי אחד: וגם, או, ולא.

const age = 19;
const hasTicket = true;
const isBlocked = false;

if (age >= 18 && hasTicket && !isBlocked) {
    console.log("אפשר להיכנס לאירוע");
}

const isWeekend = false;
const isHoliday = true;

if (isWeekend || isHoliday) {
    console.log("אין לימודים היום");
}
&& ANDכל התנאים חייבים להיות אמת. אם אחד מהם שקר, כל הביטוי שקר.
|| ORמספיק שתנאי אחד יהיה אמת. רק אם כולם שקר, כל הביטוי שקר.
! NOTהופך אמת לשקר ושקר לאמת. שימושי לבדיקות כמו “לא חסום”.

שערים לוגיים ב־JavaScript

ב־JS יש שלושה אופרטורים בוליאניים בסיסיים: &&, ||, !. מהם אפשר לבנות שערים לוגיים נוספים כמו XOR, NAND ו־NOR.

שער כתיבה ב־JS מתי יוצא true?
AND a && b רק כש־a וגם b הם true
OR a || b כש־a או b הם true
NOT !a כש־a הוא false
XOR a !== b כשבדיוק אחד מהם true
NAND !(a && b) כל מצב חוץ משניהם true
NOR !(a || b) רק כששניהם false
XNOR a === b כששניהם אותו דבר: שניהם true או שניהם false

חשוב: ב־JavaScript קיימים גם &, |, ^, ~, אבל אלה אופרטורים ביטיים. לתנאים רגילים נשתמש ב־&&, ||, !.

שילובים של שערים לוגיים

בפרויקטים אמיתיים כמעט תמיד משלבים כמה שערים יחד כדי לתאר חוק עסקי: מי יכול להיכנס, מתי להציג כפתור, או מתי פעולה מותרת.

const isLoggedIn = true;
const isAdmin = false;
const isEditor = true;
const isBlocked = false;
const hasPaid = true;

// AND + OR + NOT
const canEdit = isLoggedIn && (isAdmin || isEditor) && !isBlocked;

// XOR - בדיוק אחד מהם נכון
const hasOnlyOneRole = isAdmin !== isEditor;

// NAND - לא שניהם ביחד
const notAdminAndBlocked = !(isAdmin && isBlocked);

// NOR - אף אחד מהתנאים לא נכון
const isGuest = !(isAdmin || isEditor);

console.log("יכול לערוך:", canEdit);
console.log("יש בדיוק תפקיד אחד:", hasOnlyOneRole);
console.log("לא גם אדמין וגם חסום:", notAdminAndBlocked);
console.log("אורח:", isGuest);
console.log("יכול לקנות:", isLoggedIn && hasPaid && !isBlocked);
סוגריים משנים משמעות: a && (b || c) לא תמיד זהה ל־(a && b) || c. כשיש שילובים, כותבים סוגריים בשביל קריאות ובטיחות.

switch - תנאי עם הרבה אפשרויות

כשבודקים משתנה אחד מול הרבה ערכים קבועים, switch יכול להיות קריא יותר מרצף ארוך של else if.

const role = "admin";

switch (role) {
    case "admin":
        console.log("גישה מלאה למערכת");
        break;
    case "editor":
        console.log("אפשר לערוך תוכן");
        break;
    case "viewer":
        console.log("אפשר לצפות בלבד");
        break;
    default:
        console.log("תפקיד לא מוכר");
}
זוכרים: בלי break, הקוד ממשיך ל־case הבא.

לולאות: for, while, do...while

לולאות מריצות קוד כמה פעמים. בוחרים סוג לולאה לפי השאלה: יודעים כמה פעמים? לא יודעים מתי זה ייגמר? חייבים לרוץ לפחות פעם אחת?

for משתמשים כשבדרך כלל יודעים מראש כמה פעמים להריץ. צורת העבודה: יוצרים משתנה התחלתי, בודקים תנאי, מריצים את גוף הלולאה, מעדכנים את המשתנה וחוזרים שוב לבדיקה.
while משתמשים כשלא יודעים מראש כמה חזרות יהיו. צורת העבודה: קודם בודקים תנאי, ואם הוא אמת מריצים את גוף הלולאה. בתוך הלולאה חייבים לשנות משהו כדי שהתנאי יהפוך בסוף לשקר.
do...while משתמשים כשהקוד חייב לרוץ לפחות פעם אחת. צורת העבודה: קודם מריצים את גוף הלולאה, אחר כך בודקים תנאי, ואם הוא אמת חוזרים לעוד סיבוב. 
for (let i = 1; i <= 3; i++) {
    console.log(`for: ${i}`);
}

let count = 1;
while (count <= 3) {
    console.log(`while: ${count}`);
    count++;
}

let answer;
do {
    answer = "yes";
    console.log("do...while רץ לפחות פעם אחת");
} while (answer !== "yes");

for...of ו־for...in

for...of ו־for...in נראים דומים, אבל הם עונים על שאלות שונות: האם אני רוצה את הערכים עצמם, או את השמות של המפתחות?

for...of עובר על הערכים של מבנה שאפשר לעבור עליו לפי סדר, כמו מערך או מחרוזת. בכל סיבוב המשתנה מקבל את הערך עצמו: למשל "HTML", ואז "CSS".
for...in עובר על המפתחות של אובייקט. בכל סיבוב המשתנה מקבל את שם המפתח כסטרינג, למשל "name", ואז משתמשים ב־user[key] כדי להגיע לערך.
Object.entries() + for...of השיטה המודרנית והנוחה לאובייקטים כשצריך גם מפתח וגם ערך יחד: for (const [key, value] of Object.entries(user)).
מבנה נתונים לולאה מומלצת מה מקבלים בכל סיבוב?
מערך Array for...of את הערך עצמו: "HTML", "CSS"
אובייקט Object for...in רק מפתחות כסטרינגים: "name", "role"
אובייקט מודרני Object.entries() + for...of זוגות של מפתח וערך יחד: [key, value] 
const skills = ["HTML", "CSS", "JavaScript"];

// מתאים למערכים - מקבלים ערכים
for (const skill of skills) {
    console.log(`מיומנות: ${skill}`);
}

const user = {
    name: "נועה",
    role: "student",
    active: true
};

// מתאים לאובייקטים - מקבלים מפתחות
for (const key in user) {
    console.log(`${key}: ${user[key]}`);
}

// מודרני ונוח - מקבלים מפתח וערך יחד
for (const [key, value] of Object.entries(user)) {
    console.log(`הערך של ${key} הוא ${value}`);
}

break ו־continue

break עוצר לולאה לגמרי. continue מדלג על הסיבוב הנוכחי וממשיך לסיבוב הבא.

for (let i = 1; i <= 8; i++) {
    if (i === 3) {
        console.log("מדלגים על 3");
        continue;
    }

    if (i === 6) {
        console.log("עוצרים ב-6");
        break;
    }

    console.log(i);
}

שילוב תנאים ולולאות

רוב התרגילים האמיתיים משלבים כמה כלים: עוברים על נתונים, בודקים תנאי, ואז מחליטים מה לעשות.

const numbers = [4, 9, 12, 15, 22, 30];
let sum = 0;

for (const number of numbers) {
    if (number % 3 === 0 && number % 2 === 0) {
        console.log(`${number} מתחלק גם ב-2 וגם ב-3`);
        sum += number;
    }
}

console.log("סכום:", sum);

Nested Conditions - תנאים מקוננים

תנאי מקונן הוא תנאי בתוך תנאי. משתמשים בזה כשצריך לבדוק שלבים, אבל חשוב לא להעמיק יותר מדי כדי שהקוד יישאר קריא.

const user = {
    isLoggedIn: true,
    role: "admin",
    hasTwoFactor: true
};

if (user.isLoggedIn) {
    if (user.role === "admin") {
        if (user.hasTwoFactor) {
            console.log("כניסה מאושרת לאזור ניהול");
        } else {
            console.log("צריך אימות דו שלבי");
        }
    } else {
        console.log("אין הרשאת ניהול");
    }
} else {
    console.log("צריך להתחבר קודם");
}

מה זה פונקציה?

פונקציה = קופסה של הוראות

במקום לכתוב את אותו קוד כמה פעמים, נותנים לו שם ומפעילים אותו מתי שצריך.

function sayHello() {
    console.log("שלום כיתה!");
    console.log("ברוכים הבאים לשיעור פונקציות");
}

sayHello();
sayHello();
שםמסביר מה הפונקציה עושה
גוףהקוד שרץ כשקוראים לפונקציה
קריאהsayHello() מפעיל את הקוד

פרמטרים וארגומנטים

פרמטרים מאפשרים לפונקציה לקבל מידע מבחוץ. כך אותה פונקציה יכולה לעבוד בכל פעם עם ערכים אחרים.

function introduceStudent(name, age, grade) {
    console.log(`שם: ${name}`);
    console.log(`גיל: ${age}`);
    console.log(`ציון אחרון: ${grade}`);
}

introduceStudent("נועה", 16, 95);
introduceStudent("דניאל", 17, 88);

פרמטר הוא השם בתוך ההגדרה של הפונקציה. ארגומנט הוא הערך ששולחים בזמן הקריאה.

return - להחזיר תוצאה

לא כל פונקציה צריכה רק להדפיס. הרבה פעמים נרצה שהיא תחזיר ערך כדי שנוכל לשמור אותו, לחשב איתו או להעביר אותו לפונקציה אחרת.

function calculateVat(price) {
    return price * 0.17;
}

function finalPrice(price) {
    return price + calculateVat(price);
}

const productPrice = 100;
console.log("מע״מ:", calculateVat(productPrice));
console.log("מחיר סופי:", finalPrice(productPrice));
חשוב: קוד שנמצא אחרי return באותה פונקציה לא ירוץ.

סוגי פונקציות ב־JS

Function Declaration 
function add(a, b) {
    return a + b;
}
Function Expression 
const multiply = function(a, b) {
    return a * b;
};
Arrow Function 
const divide = (a, b) => a / b;
function add(a, b) {
    return a + b;
}

const multiply = function(a, b) {
    return a * b;
};

const divide = (a, b) => a / b;

console.log("חיבור:", add(5, 3));
console.log("כפל:", multiply(5, 3));
console.log("חילוק:", divide(12, 3));

פונקציות קצרות עם Arrow

Arrow functions נוחות במיוחד לפעולות קצרות, למשל חישובים קטנים או עבודה עם מערכים.

const double = num => num * 2;
const isAdult = age => age >= 18;
const createMessage = name => `שלום ${name}, בהצלחה בשיעור!`;

console.log(double(7));
console.log(isAdult(20));
console.log(createMessage("מאיה"));
מקרה תחביר מקוצר
פרמטר אחד num => num * 2
החזרה מיידית בלי return ובלי סוגריים מסולסלים
כמה שורות משתמשים ב־{} וב־return

פונקציה בתוך תנאים ולולאות

כשנותנים שם ללוגיקה, קל להשתמש בה שוב בתוך לולאות ותנאים.

function isEven(num) {
    return num % 2 === 0;
}

for (let i = 1; i <= 10; i++) {
    if (isEven(i)) {
        console.log(`${i} הוא מספר זוגי`);
    }
}

שאלה לכיתה

למה עדיף לקרוא לפונקציה isEven במקום לכתוב בכל מקום num % 2 === 0?

כי השם מסביר את הכוונה, ואם הלוגיקה משתנה בעתיד משנים אותה במקום אחד בלבד.

פונקציות עם מערכים

פונקציה יכולה לקבל מערך, לעבור עליו, לבנות מערך חדש ולהחזיר תוצאה נקייה.

function getPassingGrades(grades) {
    const passing = [];

    for (let i = 0; i < grades.length; i++) {
        if (grades[i] >= 60) {
            passing.push(grades[i]);
        }
    }

    return passing;
}

console.log(getPassingGrades([100, 55, 78, 42, 91]));

פונקציות עם אובייקטים

בפרויקטים אמיתיים מעבירים הרבה פעמים אובייקט שלם לפונקציה, ולא עשרה פרמטרים נפרדים.

function printUserCard(user) {
    console.log("----- כרטיס משתמש -----");
    console.log(`שם: ${user.name}`);
    console.log(`תפקיד: ${user.role}`);
    console.log(`פעיל: ${user.isActive ? "כן" : "לא"}`);
}

const user = {
    name: "אלכס",
    role: "Frontend Developer",
    isActive: true
};

printUserCard(user);

Callback - פונקציה שמקבלת פונקציה

ב־JavaScript פונקציה היא ערך. לכן אפשר לשלוח פונקציה לפונקציה אחרת, ולהחליט מבחוץ מה יקרה בפנים.

function runOperation(a, b, operation) {
    const result = operation(a, b);
    console.log(`התוצאה היא: ${result}`);
}

const add = (x, y) => x + y;
const power = (x, y) => x ** y;

runOperation(4, 3, add);
runOperation(4, 3, power);

הרעיון הזה יופיע הרבה בהמשך עם אירועים, מערכים ופעולות אסינכרוניות.

חיבור אלמנטים מ־HTML ל־JavaScript

כדי ש־JavaScript ישפיע על הדף, קודם צריך “לתפוס” אלמנט מה־HTML. בדרך כלל עושים את זה עם id, ואז אפשר לקרוא ערכים, להגיב ללחיצות ולהחליף קלאסים של CSS.

<input id="userName" value="נועה">
<button id="helloBtn">הצג ברכה</button>
<p id="message" class="message"></p>
.message.success {
    color: white;
    background: teal;
    padding: 12px;
}
const input = document.getElementById("userName");
const button = document.getElementById("helloBtn");
const message = document.getElementById("message");

button.addEventListener("click", function() {
    const name = input.value;
    message.textContent = `שלום ${name}!`;
    message.classList.add("success");
    console.log(message.textContent);
});
1. הזנת נתוניםקוראים ערך משדה עם input.value.
2. הצגה בדףכותבים תוצאה חזרה ל־HTML עם textContent.
3. החלפת קלאסיםמשנים עיצוב דרך CSS עם classList.add, remove או toggle.

Scope - איפה המשתנה קיים?

משתנה שמוגדר בתוך פונקציה חי בתוך הפונקציה. מבחוץ הוא לא מוכר.

let globalMessage = "אני גלובלי";

function showScope() {
    let localMessage = "אני מקומי";
    console.log(globalMessage);
    console.log(localMessage);
}

showScope();
console.log("מחוץ לפונקציה אפשר להשתמש ב-globalMessage");
// console.log(localMessage); // יגרום לשגיאה
כלל אצבע: משתנה צריך להיות מוגדר במקום הכי קטן שבו באמת צריכים אותו.

Global, Function ו־Block Scope

ב־JavaScript יש כמה “אזורים” שבהם משתנים חיים. ההבדל בין let, const ו־var חשוב במיוחד בתוך בלוקים ופונקציות.

const appName = "Grade App"; // Global scope

function checkGrade(grade) {
    const message = "בודקים ציון"; // Function scope
    console.log(appName);
    console.log(message);

    if (grade >= 60) {
        const status = "עבר"; // Block scope
        console.log(status);
    }

    // console.log(status); // שגיאה: status קיים רק בתוך ה-if
}

checkGrade(88);
סוג Scope איפה נוצר?
Global מחוץ לכל פונקציה או בלוק
Function בתוך פונקציה
Block בתוך {} עם let או const

Scope Chain ו־Shadowing

כשמשתנה לא נמצא באזור הקרוב, JavaScript מחפש אותו החוצה. אם יש משתנה פנימי עם אותו שם, הוא “מסתיר” את החיצוני.

const userName = "גלובלי";

function outer() {
    const userName = "בתוך outer";

    function inner() {
        const userName = "בתוך inner";
        console.log(userName);
    }

    inner();
    console.log(userName);
}

outer();
console.log(userName);

Shadowing לא תמיד טעות, אבל בכיתה נעדיף שמות ברורים כדי לא לבלבל בין משתנים באזורים שונים.

פונקציה שזוכרת משתנה - Closure בסיסי

Closure קורה כשפונקציה פנימית ממשיכה להשתמש במשתנים של פונקציה חיצונית, גם אחרי שהפונקציה החיצונית סיימה לרוץ.

function createCounter() {
    let count = 0;

    return function() {
        count++;
        console.log(`הספירה עכשיו: ${count}`);
    };
}

const counter = createCounter();
counter();
counter();
counter();
לא צריך לשלוט בזה לעומק עדיין: מספיק להבין שפונקציות יכולות “לזכור” משתנים מהסביבה שבה הן נוצרו.

תרגול מסכם בלייב

נבנה פונקציות קטנות, ואז נחבר אותן לפתרון אחד.

function getAverage(numbers) {
    let sum = 0;

    for (let i = 0; i < numbers.length; i++) {
        sum += numbers[i];
    }

    return sum / numbers.length;
}

function getStatus(average) {
    if (average >= 90) return "מצוין";
    if (average >= 75) return "טוב מאוד";
    if (average >= 60) return "עבר";
    return "צריך תרגול נוסף";
}

const grades = [80, 92, 71, 88];
const average = getAverage(grades);

console.log("ממוצע:", average);
console.log("סטטוס:", getStatus(average));

הצגת ערכים חזרה בתוך ה־HTML

אחרי שקוד JS מחשב ערך, אפשר להציג אותו בדף בעזרת textContent, innerHTML או שינוי value של שדה.

<input id="nameInput" value="נועה">
<p id="message"></p>
const input = document.getElementById("nameInput");
const message = document.getElementById("message");

message.textContent = `שלום ${input.value}, ברוכה הבאה!`;

טיפ: כשמציגים טקסט רגיל נעדיף textContent. את innerHTML שומרים למקרים שבהם באמת צריך להכניס HTML.

שינוי עיצוב וקלאסים של אלמנטים

JavaScript יכול לשנות איך אלמנט נראה: או דרך style לשינוי ישיר, או דרך classList כדי להוסיף ולהוריד קלאסים מוגדרים ב־CSS.

<div id="card" class="card">כרטיס משתמש</div>
.card.active {
    background: orange;
    color: white;
}
const card = document.getElementById("card");

card.style.border = "3px solid teal";
card.classList.add("active");
card.classList.toggle("highlight");

console.log("העיצוב והקלאסים עודכנו");

בפרויקטים אמיתיים עדיף לרוב לשנות קלאסים עם classList, כי העיצוב נשאר מסודר בקובץ CSS.

דוגמה קצרה: מחשבון HTML + JS

בדוגמה הזו ה־HTML נותן שדות וכפתורים, וה־JavaScript קורא את הערכים, מחשב ומחזיר את התוצאה לתוך הדף.

<input id="num1" type="number" value="10">
<input id="num2" type="number" value="5">

<button onclick="calculate('+')">+</button>
<button onclick="calculate('-')">-</button>
<button onclick="calculate('*')">*</button>
<button onclick="calculate('/')">/</button>

<p id="result">התוצאה תופיע כאן</p>
function calculate(operator) {
    const num1 = Number(document.getElementById("num1").value);
    const num2 = Number(document.getElementById("num2").value);
    const result = document.getElementById("result");
    let answer;

    if (operator === "+") answer = num1 + num2;
    else if (operator === "-") answer = num1 - num2;
    else if (operator === "*") answer = num1 * num2;
    else if (operator === "/") answer = num2 !== 0 ? num1 / num2 : "אי אפשר לחלק באפס";

    result.textContent = `תוצאה: ${answer}`;
    console.log(result.textContent);
}

שיעורי בית

1. בדיקת הרשאות

כתבו תנאי שמשתמש ב־if, &&, || ו־! כדי לבדוק אם משתמש יכול להיכנס למערכת.

2. תפריט עם switch

כתבו switch שמקבל פעולה כמו "save", "delete", "edit" ומדפיס הודעה מתאימה.

3. סינון מספרים

כתבו לולאה שמדפיסה רק מספרים זוגיים שמתחלקים גם ב־3. השתמשו ב־continue כדי לדלג על מספרים לא מתאימים.

4. מעבר על אובייקט

צרו אובייקט משתמש והדפיסו את כל המפתחות והערכים שלו בעזרת for...in.

5. מחשבון קטן

כתבו פונקציות add, subtract, multiply, divide שמחזירות תוצאה ומדפיסות אותה לקונסול.

6. בדיקת משתמש

כתבו פונקציה שמקבלת אובייקט משתמש ומחזירה אם הוא יכול להיכנס: גיל 18+ וגם isActive === true.

7. מערך ציונים

כתבו פונקציה שמקבלת מערך ציונים ומחזירה אובייקט עם ממוצע, ציון גבוה וציון נמוך.

8. Callback

כתבו פונקציה שמקבלת מערך ופונקציית בדיקה, ומדפיסה רק את הערכים שעוברים את הבדיקה.

סיכום

  • תנאים מאפשרים לקוד לבחור מסלול בעזרת if, else if, else ו־switch.
  • לוגיקה בוליאנית מחברת תנאים עם &&, || ו־!, ומהם בונים גם XOR, NAND, NOR ו־XNOR.
  • לולאות מאפשרות לחזור על פעולה עם for, while, do...while, for...of ו־for...in.
  • break עוצר לולאה ו־continue מדלג לסיבוב הבא.
  • פונקציה נותנת שם לקוד שאפשר להפעיל שוב.
  • פרמטרים מאפשרים לשלוח מידע לפונקציה.
  • return מחזיר תוצאה להמשך שימוש.
  • יש כמה דרכים להגדיר פונקציות: Declaration, Expression, Arrow.
  • פונקציות משתלבות עם תנאים, לולאות, מערכים, אובייקטים ו־Callbacks.
  • Scope קובע איפה משתנים ופונקציות קיימים ואיפה אפשר להשתמש בהם.
  • DOM מחבר בין HTML ל־JavaScript: תופסים אלמנטים, קוראים ערכים, מציגים תוצאות ומשנים עיצוב וקלאסים.
